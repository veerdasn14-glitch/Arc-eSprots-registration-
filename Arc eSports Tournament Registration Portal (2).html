function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    
    // Auto-create header row if sheet is fresh
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Slot", "Player Name", "eFootball IGN", "Playstyle", "Entry Fee", "12-Digit UTR", "Status"]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#0A0D14").setFontColor("#00F2FE");
    }
    
    sheet.appendRow([
      timestamp,
      data.slot,
      data.name,
      data.ign,
      data.playstyle,
      data.fee,
      "'" + data.utr,
      "Pending Verification"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", slot: data.slot }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
